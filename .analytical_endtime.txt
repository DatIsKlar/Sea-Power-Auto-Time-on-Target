
        // Beta-only: pure analytical estimate using CalculateMissileInterceptTimeAnalytical.
        // 2-pass closed-form solver (no trajectory integration). Test alongside BuildFlyoutEndTime.
        private static float AnalyticalEndTime(ObjectBase unit, AmmunitionParameters ap, ObjectBase target)
        {
            if (_calcInterceptAnalyticalMethod == null) return -1f;
            try
            {
                Vector3 launchPos = unit.transform.position;
                Vector3 targetPos = target.transform.position;
                Vector3 targetVel = target._velocityVecInUnity;

                float dragFactor = -1f;
                if (_getDragFactorMethod != null)
                    dragFactor = (float)_getDragFactorMethod.Invoke(ap, new object[] { unit.IsAirUnit });

                object result = _calcInterceptAnalyticalMethod.Invoke(null, new object[]
                {
                    ap, launchPos, unit._velocityInKnots, targetPos, targetVel, dragFactor
                });
                if (result == null) return -1f;

                var tuple = (ValueTuple<float, bool, Vector3>)result;
                if (!tuple.Item2) return -1f;
                return tuple.Item1 > 0.01f ? tuple.Item1 : -1f;
            }
            catch
            {
                return -1f;
            }
        }
