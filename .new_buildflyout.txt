        private static float BuildFlyoutEndTime(ObjectBase unit, AmmunitionParameters ap, ObjectBase target)
        {
            if (_buildFlyoutMethod == null) return -1f;
            try
            {
                Vector3 launchPos = unit.transform.position;
                float launchAlt = Mathf.Max(launchPos.y, 0f);
                Vector3 targetPos = target.transform.position;
                Vector3 targetVel = target._velocityVecInUnity;
                float flatDist = launchPos.FlatDistance(targetPos);
                if (flatDist < 0.001f) return -1f;

                float dragFactor = -1f;
                if (_getDragFactorMethod != null)
                    dragFactor = (float)_getDragFactorMethod.Invoke(ap, new object[] { unit.IsAirUnit });

                float lifetime = ap._maxFlightTime > 0f ? ap._maxFlightTime : 600f;

                // Apply evasive target velocity modification (matches EstimateShot lines 1222-1229)
                float targetVelMag = targetVel.magnitude;
                if (targetVelMag > 0f)
                {
                    Vector3 radial = targetPos - launchPos;
                    radial.y = 0f;
                    if (radial.magnitude > 0.001f)
                    {
                        targetVel += radial.normalized * (targetVelMag * 0.8f);
                        float newMag = targetVel.magnitude;
                        targetVel = targetVel.normalized * Mathf.Min(newMag, targetVelMag);
                    }
                }

                const float KU = 0.0076554087f;

                // 3-fallback chain for seed time (matches EstimateShot lines 1250-1258)
                float estTime = -1f;
                if (_calcInterceptAnalyticalMethod != null)
                {
                    object result = _calcInterceptAnalyticalMethod.Invoke(null, new object[]
                    {
                        ap, launchPos, unit._velocityInKnots, targetPos, targetVel, dragFactor
                    });
                    if (result != null)
                    {
                        var tuple = (ValueTuple<float, bool, Vector3>)result;
                        estTime = tuple.Item1;
                    }
                }
                if (estTime < 0f && _calcInterceptSimpleMethod != null)
                {
                    object result = _calcInterceptSimpleMethod.Invoke(null, new object[]
                    {
                        ap, launchPos, targetPos, targetVel
                    });
                    if (result != null) estTime = (float)result;
                }
                if (estTime < 0f)
                {
                    estTime = flatDist / Mathf.Max(ap._maxVelocityInKnots * KU, 1f);
                }
                estTime = Mathf.Clamp(estTime, 0f, Mathf.Min(lifetime, 7200f));

                // Predict intercept position and compute flatDistTotal
                Vector3 predicted = targetPos + targetVel * estTime;
                float flatDistTotal = Mathf.Max(launchPos.FlatDistance(predicted), 1f);
                float targetAlt = Mathf.Max(predicted.y, 0f);

                // Loft search (matches EstimateShot lines 1263-1272)
                float loftAltOverride = -1f;
                if (_loftCapMethod != null && _searchOptimalLoftMethod != null && _kinematicsField != null)
                {
                    object kinematicsObj = _kinematicsField.GetValue(ap);
                    int kinematics = kinematicsObj != null ? (int)kinematicsObj : 0;
                    if (kinematics != 0)
                    {
                        object loftCapResult = _loftCapMethod.Invoke(null, new object[] { ap, launchAlt, targetAlt });
                        if (loftCapResult != null)
                        {
                            float loftCap = (float)loftCapResult;
                            float altFloor = Mathf.Max(launchAlt, targetAlt);
                            if (loftCap > altFloor + 0.5f)
                            {
                                object loftResult = _searchOptimalLoftMethod.Invoke(null, new object[]
                                {
                                    ap, launchPos, targetPos, targetVel, unit._velocityInKnots,
                                    unit.IsAirUnit, dragFactor, altFloor, loftCap
                                });
                                if (loftResult != null) loftAltOverride = (float)loftResult;
                            }
                        }
                    }
                }

                object profile = _buildFlyoutMethod.Invoke(null, new object[]
                {
                    ap, launchAlt, targetAlt, flatDistTotal,
                    unit._velocityInKnots, unit.IsAirUnit,
                    dragFactor, loftAltOverride, lifetime
                });
                if (profile == null) return -1f;

                if (_endTimeProp == null)
                    _endTimeProp = profile.GetType().GetProperty("EndTime");
                if (_endTimeProp == null) return -1f;
                float endTime = (float)_endTimeProp.GetValue(profile);
                if (endTime <= 0.01f) return -1f;

                if (_flatDistAtMethod == null)
                    _flatDistAtMethod = profile.GetType().GetMethod("FlatDistAt", new Type[] { typeof(float) });
                if (_flatDistAtMethod == null) return -1f;

                // Use the MODIFIED targetVel for the Gap function (matches EstimateShot line 1287)
                Vector3 relFlat = new Vector3(targetPos.x - launchPos.x, 0f, targetPos.z - launchPos.z);
                Vector3 velFlat = new Vector3(targetVel.x, 0f, targetVel.z);

                float Gap(float time)
                {
                    float targetDist = (relFlat + velFlat * time).magnitude;
                    float missileDist = (float)_flatDistAtMethod.Invoke(profile, new object[] { time });
                    return targetDist - missileDist;
                }

                // Scan 48 segments for zero-crossing, then binary-search 40 iterations
                float interceptTime = -1f;
                float prevT = 0f;
                if (Gap(0f) <= 0f)
                {
                    interceptTime = 0f;
                }
                else
                {
                    for (int i = 1; i <= 48; i++)
                    {
                        float t = endTime * (float)i / 48f;
                        float g = Gap(t);
                        if (g <= 0f)
                        {
                            float lo = prevT, hi = t;
                            for (int j = 0; j < 40; j++)
                            {
                                float mid = (lo + hi) * 0.5f;
                                if (Gap(mid) > 0f) lo = mid;
                                else hi = mid;
                            }
                            interceptTime = (lo + hi) * 0.5f;
                            break;
                        }
                        prevT = t;
                    }
                }

                // MinVelocity post-check (matches EstimateShot lines 1326-1330)
                if (interceptTime >= 0f && _kinematicsField != null)
                {
                    object kinematicsObj = _kinematicsField.GetValue(ap);
                    int kinematics = kinematicsObj != null ? (int)kinematicsObj : 0;
                    if (kinematics != 0)
                    {
                        if (_speedUnityAtMethod == null)
                            _speedUnityAtMethod = profile.GetType().GetMethod("SpeedUnityAt", new Type[] { typeof(float) });
                        if (_speedUnityAtMethod != null)
                        {
                            object speedResult = _speedUnityAtMethod.Invoke(profile, new object[] { interceptTime });
                            if (speedResult != null)
                            {
                                float speedUnity = (float)speedResult;
                                float minSpeedUnity = ap.MinVelocity * 1.1f * KU;
                                if (speedUnity < minSpeedUnity) interceptTime = -1f;
                            }
                        }
                    }
                }

                return interceptTime > 0.01f ? interceptTime : -1f;
            }
            catch
            {
                return -1f;
            }
        }
